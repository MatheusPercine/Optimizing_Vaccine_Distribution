\begin{simbolos}
\item[$C_{ij}, C_{ik}, C_{kj}, C_{jp}$] Custo Total de transporte (em horas) entre os respectivos nós
\item[$C_{ij}^{rod}, C_{ij}^{aereo}$] Custo Total do transporte por modal (rodoviário ou aéreo)
\item[$d_{jv}$] Demanda total de doses da vacina do tipo $v$ no município $j$
\item[$e_{iv}, e_{jv}, e_{kv}$] Tempo restante até a expiração da vacina do tipo $v$ no nó de origem $i$, destino $j$ ou centro $k$
\item[$f_{pv}$] Quantidade faltante (não atendida) da vacina do tipo $v$ no posto $p$
\item[$I$] Conjunto de municípios de origem
\item[$J$] Conjunto de municípios de destino
\item[$K$] Conjunto de centros estaduais
\item[$L$] Limite superior para o tempo total de distribuição das vacinas (tempo de gargalo)
\item[$M$] Constante de penalização de demanda não atendida
\item[$o_{iv}$] Quantidade de vacinas do tipo $v$ ofertadas no município $i$
\item[$P_j$] Conjunto de postos de saúde pertencentes ao município $j$
\item[$Pop_j$] População do município $j$
\item[$q_{jp}$] Capacidade máxima de transporte entre o município $j$ e o posto $p$
\item[$r_{pv}$] Demanda da vacina do tipo $v$ no posto de destino $p$
\item[$R_0$] Taxa de transmissão básica do vírus (Trabalho correlato)
\item[$S_{AB}$] Economia (saving) gerada pela fusão das rotas para A e B (Trabalho correlato)
\item[$U^{rod}, U^{aereo}$] Capacidade média de carga de um veículo do modal rodoviário ou aéreo
\item[$u_{ij}, u_{ik}, u_{kj}$] Capacidade efetiva agregada de transporte entre os respectivos nós
\item[$V$] Conjunto de tipos de vacina
\item[$W_i^{rod}, W_i^{aereo}$] Quantidade de veículos disponíveis no município $i$ para cada modal
\item[$x_{ijv}$] Quantidade de vacinas do tipo $v$ transportadas do município $i$ para o município $j$
\item[$x_{ikv}$] Quantidade de vacinas do tipo $v$ transportadas do município $i$ para o centro estadual $k$
\item[$x_{ij}^s$] Variável binária de roteamento de veículos (Trabalho correlato)
\item[$y_{kjv}$] Quantidade de vacinas do tipo $v$ transportadas do centro estadual $k$ para o município $j$
\item[$z_{jpv}$] Quantidade de vacinas do tipo $v$ transportadas do município $j$ para seu posto de saúde $p$

\item[$\alpha$] Fator de penalização associado ao transporte aéreo
\item[$\beta$] Fator de tortuosidade da distância geodésica
\item[$\gamma$] Fração da rede de vacinação capturada pelos postos considerados
\item[$\Delta_{ab}$] Distância geodésica entre as localidades $a$ e $b$
\item[$\Delta_{ab}^{real}$] Distância real estimada para o deslocamento rodoviário entre $a$ e $b$
\item[$\delta^{rod}, \delta^{aereo}$] Tempo de processamento logístico do modal rodoviário ou aéreo
\item[$\theta_{rod}, \theta_{aereo}$] Velocidade média do transporte rodoviário ou aéreo
\item[$\rho_{iv}, \rho_{jv}$] Fator de penalidade associado ao tempo restante de validade da vacina
\item[$\tau_{ij}^{rod}, \tau_{ij}^{aereo}$] Tempo de deslocamento do modal rodoviário ou aéreo entre $i$ e $j$
\item[$\phi_v$] Vida útil típica (em horas) da vacina $v$

\item[$\sum$] Operador de somatório
\item[$\in$] Pertence a um conjunto
\item[$\mathbb{R}_{\ge 0}$] Conjunto dos números reais não negativos
\item[$\mathbb{Z}_{\ge 0}$] Conjunto dos números inteiros não negativos
\end{simbolos}
