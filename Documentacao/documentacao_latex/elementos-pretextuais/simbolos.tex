\begin{simbolos}
\item[$x_{ijt}$] Quantidade de vacinas transportadas do município $i$ para o município $j$ do tipo $t$
\item[$z_{abt}$] Quantidade de vacinas transportadas do ponto $a$ para o ponto $b$ do tipo $t$
\item[$x_{ikt}$] Quantidade de vacinas transportadas do município $i$ para o centro estadual $k$
\item[$y_{kjt}$] Quantidade de vacinas transportadas do centro estadual $k$ para o município $j$

\item[$c_{ij}$] Custo (tempo) de transporte entre os municípios $i$ e $j$
\item[$c_{ik}, c_{kj}$] Custo (tempo) de transporte nas etapas intermunicipais via centro estadual
\item[$p_{it}$] Penalidade associada ao tempo de expiração da vacina $t$ no município $i$

\item[$o_{it}$] Oferta de vacinas do tipo $t$ no município $i$
\item[$d_{jt}$] Demanda de vacinas do tipo $t$ no município $j$
\item[$u_{ij}$] Capacidade máxima de transporte entre os municípios $i$ e $j$

\item [$\tau_{ij}^{rod}$] é o tempo de deslocamento do modal rodoviário entre o município de origem $i$ e o município de destino $j$.
\item [$\tau_{ij}^{areo}$] é o tempo de deslocamento do modal aéreo entre o município de origem $i$ e o município de destino $j$.
\item [$\tau_{ab}$] é o tempo de deslocamento do modal rodoviário entre o ponto de origem de origem $a$ até o ponto de destino $b$.

\item[$\alpha$] Tempo de preparação do transporte

\item[$\delta^{rod}$] Tempo de processamento logístico do modal rodoviário, no destino
\item[$\delta^{areo}$] Tempo de processamento logístico do modal aéreo, no destino

\item[$I, J$] Conjuntos de municípios
\item[$K$] Conjunto de centros estaduais
\item[$T$] Conjunto de tipos de vacina
\item[$A, B$] Conjuntos de pontos de origem e destino no modelo intramunicipal

\item[$\sum$] Operador de somatório
\item[$\in$] Pertence a um conjunto
\item[$\mathbb{Z}_{\geq 0}$] Conjunto dos números inteiros não negativos
\end{simbolos}
